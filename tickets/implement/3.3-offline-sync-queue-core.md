// reachability.ts
/** Whether the cohorts responsible for these blocks can be reached now — the gate of
 * docs/offline-sync.md §When to submit. db-p2p implements it over the key network (stage 6);
 * absent, the queue treats the cohort as always reachable, which is right for TestTransactor,
 * the local transactor and the in-process mesh. Asked with each collection's header id and its
 * last known log tail id, so an implementation resolves cohorts from routing keys alone and
 * never needs the network to answer whether the network can be reached. */
export interface ICohortReachability {
	reachable(blockIds: readonly BlockId[]): Promise<boolean>;
}

// queue.ts
export type OfflineQueueOptions = {
	reachability?: ICohortReachability;
	/** An external wake signal (db-p2p's connection events in stage 6): subscribe the queue's
	 * `wake` and return the unsubscribe, which `dispose` calls. */
	wakeOn?: (wake: () => void) => () => void;
	/** Fail an entry that stays unsettled this long, with reason `expired`. Default: no limit. */
	expireAfterMs?: number;
	/** The re-check timer while the cohort is unreachable or after a transient failure:
	 * exponential from `baseMs` to `maxMs`, jittered (`jitteredBackoffMs`). Defaults 1 s / 60 s. */
	recheck?: { baseMs?: number; maxMs?: number; rand?: RandFn };
	/** Retry bounds handed to each submission (`commitHead` / `commitEntry`). */
	submit?: Pick<SyncOptions, 'maxAttempts' | 'deadlineMs' | 'baseBackoffMs' | 'maxBackoffMs' | 'maxStalledAttempts'>;
	/** Write-ahead hook: called with the attempt(s) about to be sent, before they are sent, and
	 * awaited; a throw keeps the entry queued, nothing sent (the journal's seam, stage 5). */
	recordAttempt?: (entryId: EntryId, attempts: ReadonlyMap<CollectionId, InFlightAttempt>) => Promise<void>;
	/** Clock, for tests. */
	now?: () => number;
};

export class OfflineQueue {
	constructor(transactor: ITransactor, options?: OfflineQueueOptions);
	/** Seal each collection's open segment under a new entry and queue it. `undefined` when no
	 * collection had anything staged — nothing to queue, nothing sealed. Emits `queued`
	 * SYNCHRONOUSLY before returning, so a caller can attribute it to the commit it is in. */
	enqueue(collections: readonly Collection<any>[], init?: { id?: EntryId; timestamp?: number }): QueuedEntry | undefined;
	/** The collection callback: pass as `CollectionInitOptions.onSealedRefused` of every
	 * collection whose entries this queue drains. Never calls a latched method synchronously. */
	readonly onSealedRefused: (entryId: EntryId, error: ActionRefusedError) => void;
	entries(): readonly QueuedEntry[];          // queue order
	status(entryId: EntryId): EntryStatus | undefined;
	/** Withdraw a never-sent entry: it fails with reason `discarded` and cascades like any
	 * failure. Throws `EntryNotDiscardableError` for an entry that is in doubt (sent at least
	 * once) or being submitted, `UnknownEntryError` for an id not queued. */
	discard(entryId: EntryId): void;
	onStatus(listener: (status: EntryStatus) => void): () => void;
	onQueueState(listener: (state: OfflineQueueState) => void): () => void;
	queueState(): OfflineQueueState;
	/** Kick the drain: a new local commit (enqueue calls it), a connection event, a test. */
	wake(): void;
	/** Resolves when the queue is empty; rejects with `QueueDisposedError` on dispose. */
	whenDrained(): Promise<void>;
	/** Stop draining. Resolves once any submission in flight has been applied locally. */
	dispose(): Promise<void>;
}

Errors: `EntryNotDiscardableError`, `UnknownEntryError`, `QueueDisposedError`, each a named class in `status.ts`. `committed.durability` for a one-collection entry wraps `commitHead`'s answer with `transactionDurability` (a collection with an `undefined` class when the answer was `undefined`, `overall` then absent — the "saved, holder unknown" shape the coordinator already uses).

## Entries

An entry is `{ id, seq, queuedAt, expiresAt?, collections: Collection[], actionId, attempts, inDoubt, retained: Map<CollectionId, InFlightAttempt>, state }`. `seq` is per queue, monotonic. `id` is the caller's (the SQL layer's `StampId()`) or minted like a sync's action id (`randomBytes(16)` base64url). `actionId` is minted the same way at enqueue.

**One action id for the entry's whole life, with one exception.** Storage treats a re-send of the same action at the same revision as finishing it and at a new revision as a new write, so re-driving under the same id is always safe and keeps the member-side slot hold the write earned by losing. The exception is a `torn-not-saved` outcome (`TornActionError` with `final: true`): a re-drive under that id would find its own stranded log entry on every refresh, re-send it, and be told again that a rival built past it — a loop. A fresh id lets the next refresh treat the stranded entry as a stranger's and move on, which is exactly the proof `final: true` gives. The doc's "a refused pend" also mints a fresh id; this stage does not (the stage-1 review recorded why a refused pend proves nothing about the cycle's earlier attempts), and the next ticket corrects the doc. `retained` is cleared with the id.

**In doubt** is sticky: set the first time the queue's `recordAttempt` wrapper runs for the entry (the only moment before a send the queue is told about), cleared only by a terminal outcome or by `torn-not-saved`. While in doubt the entry is neither discardable nor expirable, and is re-driven only under its action id with its retained attempts as `resume`. A cycle's own refreshes resolve its earlier attempts (each refresh finds a landed own entry, and an attempt at an unchanged revision finishes an earlier one as the same action), so the last cycle's classification speaks for the cycle; a coordinator lagging its cohort can still hide a minority-held tail, and the sticky rule is what keeps that from ever discarding a possibly-landed entry. Record this reasoning once, as a `NOTE:` where `inDoubt` is set.

## The drain

One loop per queue, started lazily by the first `enqueue`, serialised: at most one submission in flight. Each iteration:

1. **Pending refusals.** Apply every refusal `onSealedRefused` has reported since the last iteration (see *Removing an entry*). A refusal for the entry currently submitting is left to that submission's outcome: its cycle re-checks the recorded refusal and throws it.
2. **Expiry.** Every entry whose `expiresAt` has passed and that is not in doubt fails with `expired`, in queue order, head or not. An in-doubt entry past its expiry is logged (`offline:expiry-deferred`) and kept; it expires when its doubt resolves without committing. Test plan: "`expireAfterMs` reached while unreachable".
3. **Empty queue:** resolve `whenDrained` waiters, publish state, wait for a wake.
4. **Head.** The oldest entry. If any of its collections records a refusal for it (`sealedRefusal`), fail it now (a refresh found it before the drain did — reported at once, per §Replay with a per-segment refusal). Every later entry is `queued` with `waitingFor: 'earlier-entry'`.
5. **Gate.** `reachability.reachable(ids)` with each of the head's collections' header id and its known log tail id (add `Collection.knownLogTailId(): BlockId | undefined`, the `logTailId` field). Unreachable: head `waitingFor: 'cohort'`, state `waitingFor: 'cohort'`, arm the re-check timer (unref'd, `armUnrefTimer`) at the current backoff step, wait for a wake or the timer. A wake resets the backoff step.
6. **Submit.** `attempts++`, status `submitting`, state `draining: true`. One collection: `collection.commitHead(id, { actionId, timestamp: queuedAt, resume, recordAttempt, ...submit })`. Several: a `TransactionCoordinator` over exactly the entry's collections with `pendValidation: 'none'`, `commitEntry({ entryId, actionId, timestamp, collectionIds, resume, recordAttempt }, submit)`. The queue's `recordAttempt` wrapper marks the entry in doubt, stores the attempt(s) in `retained`, then awaits the option's hook; a throw there is logged (`offline:record-attempt-failed`) and treated as a transient failure (step 7, `backoff`).
7. **Outcome.** Success: remove the entry, emit `committed`, clear `retained`, continue. Failure: `classifyCommitFailure(error)` and apply the table:

| class | transition |
|---|---|
| `cohort-unreachable` | `queued`, `waitingFor: 'cohort'`; back to step 5 with the timer armed |
| `contention`, `expired`, `unknown` | `queued`, `waitingFor: 'backoff'`, `lastError`; re-drive after the backoff timer or a wake; `unknown` is also logged (`offline:unclassified-failure entry=… error=…`) and counted in `unclassifiedFailures` |
| `guard-refused` | `failed`, reason from the class (`constraint` / `concurrent-modification`) |
| `rejected` | `failed`, `rejected` |
| `too-large` | `failed`, `too-large` |
| `torn-not-saved` | `queued`, `waitingFor: 'backoff'`; doubt cleared, fresh action id, `retained` cleared |
| `in-doubt` | `queued`, `waitingFor: 'backoff'`; stays in doubt |
| `partial` | next ticket; until then treated as `unknown` (in doubt, re-driven under the same id — the committed participants' heads are already consumed, so the re-drive commits what is left) |

A `SealedSegmentsError` out of a submission is a queue defect (the head alignment invariant broke); throw it out of the loop, which stops the drain and rejects `whenDrained`. An `ActionRefusedError` thrown directly (the head's recorded refusal) classifies as `guard-refused` when it is a `TreeGuardRefusedError`; another `ActionRefusedError` subclass is `constraint` — extract the classifier's tree-specific `refusal` mapping into one shared function and use it here and in the refusal callback, so the two never disagree.

The backoff and re-check timers share one implementation: a step counter, `jitteredBackoffMs(step, { baseMs, capMs }, rand)`, reset by a wake or a successful submission.

## Removing an entry

`fail(entry, reason, error, causedBy?)`: set the entry failed, remove it from the list, then for each of its collections `await collection.dropSealed(entry.id)` (a collection in which the entry had no segment is skipped — `sealedEntries` says). Each drop replays and may report refusals of later entries through `onSealedRefused`, which records `(entryId, error)` with `causedBy = entry.id` into the pending list — never acting synchronously, since the callback runs under the collection latch. When every drop is done, emit `failed` with `changedCollections` = the collections the entry had segments in, then process the pending refusals in queue order, each through `fail` again with its `causedBy`, until none are pending. A refusal reported for an entry already removed is ignored. A refusal reported for the entry under submission is noted and left to the submission (step 1 above), except that `discard` and expiry never touch a submitting entry either.

A refusal reported by a reader's refresh with no failure in progress (a rival's commit refused a queued entry) goes through the same pending list with no `causedBy`, and its reason comes from the shared refusal mapping.

## Disposal

`dispose()`: unsubscribe `wakeOn`, clear timers, set disposed; if a submission is in flight, abort its `SyncOptions.signal` (the queue owns an `AbortController` per submission) so the cycle stops after its current network `await`, and wait for the loop to apply that submission's outcome locally — a committed outcome is still emitted, a failure classified as above is recorded on the entry but no further entry is submitted. Entries left queued stay sealed in their collections (they are the application's; nothing is dropped). `whenDrained` rejects with `QueueDisposedError`; later `enqueue` throws it.

## Edge cases & interactions

Each names how it is verified.

- **Lifecycle, happy path** — two entries over one tree drained in order: `queued` → `submitting` → `committed` once each, in order; each log entry under its entry's action id with `timestamp = queuedAt`; a fresh handle reads both rows. **Test.**
- **Multi-collection entry** — one entry over two trees goes through `commitEntry`: one action id in both logs. **Test** (one case; the slice's own behaviour is pinned by its ticket).
- **Gate closed** — a reachability double answering `false`: nothing is sent, head `waitingFor: 'cohort'`; flipping it and calling `wake()` drains; with `recheck.baseMs` small and no wake, the timer drains it. **Test.**
- **Cascade three deep** — E1 inserts key k, E2 updates k (an `unchanged` guard on E1's image), E3 updates k again; a rival takes k before the drain: E1 fails `constraint` with `changedCollections` naming the tree, E2 fails `concurrent-modification` with `causedBy: E1`, E3 with `causedBy: E2`; the local read no longer shows any of them; the queue is empty and `whenDrained` resolves. **Test.**
- **Refusal found by a reader before the drain** — the rival commits, a reader's `update()` on the tree reports the refusal; the entry is failed before the gate opens (gate double closed); nothing sent. **Test.**
- **Transient classes** — `RefusesPendsAsShortfall`: `queued`, `waitingFor: 'cohort'`, re-driven after the double is switched off, then committed; a double that throws a plain `Error` from `commit`: `unclassifiedFailures` is 1, the re-drive uses the SAME action id (compare the recorded attempts' action ids through the transactor double's recorded requests), then commits. **Test.**
- **Crash-free in-doubt resolution** — `TailLandsButReportsStale` with `submit.maxAttempts: 1`: the first drive leaves the entry `queued` and `inDoubt: true`; `discard` throws `EntryNotDiscardableError`; an `expireAfterMs` already passed does not fail it (logged deferred); switching the double off, the next drive resumes with the retained attempt and commits; the log holds the entry once. **Test.**
- **`discard` racing the drain** — with the commit parked (`GatedCommitTransactor`), `discard` of the head throws `EntryNotDiscardableError`; `discard` of a never-sent second entry succeeds meanwhile and its segment is gone after the drop (the head's segment is in flight; `dropSealed` of a non-head is allowed). **Test** (one case, both arms).
- **A local commit seals a new segment while the drain is on the network** — stage and `enqueue` a second entry while the first's commit is parked: both commit, in order, and the second's `queued` status said `waitingFor: 'earlier-entry'`. **Test.**
- **Disposal while an attempt is in flight** — parked commit; `dispose()` resolves only after the parked commit's outcome is applied; `committed` was emitted; `whenDrained` rejected `QueueDisposedError`; the second entry was never submitted and is still sealed. **Test.**
- **`enqueue` with nothing staged** returns `undefined` and seals nothing; **`enqueue` of a collection whose open segment is empty but a sibling's is not** seals only the sibling. By inspection of `sealPending`'s `false` answer; one assertion inside the happy-path test.
- **A refusal callback for an entry the queue does not know** (another queue's, or already removed): ignored, logged once. By inspection.
- **The `onSealedRefused` callback must never call a latched method synchronously** (`CollectionInitOptions` says why): it only pushes to the pending list and calls `wake()`. By inspection; assert at the seam with a comment.
- **Status listeners that throw** are isolated and logged (`offline:listener-threw`), never allowed to break the drain. By inspection.
- **Timers** are unref'd (`armUnrefTimer`) so an idle queue does not hold a process open; cleared on dispose. By inspection.
